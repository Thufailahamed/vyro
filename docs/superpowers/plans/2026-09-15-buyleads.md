# BuyLeads Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Daily email digest at 07:00 Asia/Colombo that surfaces new RFQs to suppliers whose subscribed category IDs intersect the RFQ's line-item categories. Top 10 per supplier, no caps, no dedup. Settings page lets suppliers opt-in + multi-select categories.

**Architecture:** New `buyLeads/` API module. Cron handler in `apps/api/src/cron/buyLeads.ts` walks enabled subscriptions, derives each RFQ's category set from `rfq_items → products.categoryId`, intersects with subscription categories, picks top 10 by recency, builds email payload, sends via `NOTIFICATIONS_QUEUE` with new kind `buyleads_digest`. Queue consumer at `apps/api/src/queue/notifications.ts` is extended with a new branch for that kind. Settings endpoint mirrors the `repeatOffers/supplierRoutes.ts` pattern (`?supplierId=` query + `requireSupplierRole`). Web adds a `BuyLeadsSettingsSection` to the existing supplier `SettingsPage`.

**Tech Stack:** Hono (Workers), Drizzle ORM (D1/SQLite), Zod validation (`@vyro/validation`), Vitest with `node:sqlite` D1 shim, TanStack Query + `api.get/put` (web).

## Global Constraints

- Digest cadence: daily at 07:00 Asia/Colombo. Cron expression: `0 7 * * *` (UTC equivalent of 07:00 IST is `0 1 * * *` UTC during summer; use `0 1 * * *` UTC consistently, accept ±1h drift around DST — Lanka doesn't observe DST so `0 1:30 * * *` is safer). Final: `0 1 * * *` UTC = 07:00 Colombo year-round.
- Top N per supplier: 10, ordered by `rfqs.created_at DESC`.
- No dedup across days (re-emit RFQs even if already digested yesterday).
- No per-supplier caps. Unlimited free.
- Flag `BUYLEADS_ENABLED` in D1 `feature_flags` config section; default `false`; read via `isFeatureEnabled(d1, flag)`.
- Categories: derive from `rfq_items.productId → products.categoryId`. Match if any line-item category is in sub.categoryIds.
- Email payload built in cron handler, sent via `env.NOTIFICATIONS_QUEUE.send({ kind: 'buyleads_digest', ... })`. Consumer reads `kind` and dispatches to a new branch.
- Subscription table: 1 row per supplier. `categoryIds` stored as JSON array (text column). No junction table at MVP.
- Settings endpoint uses PUT (NOT PATCH). Supplier role check via `requireSupplierRole(ctx, supplierId, ['owner','sales','operations'])` — same pattern as `repeatOffers/supplierRoutes.ts`. `supplierId` from query string.
- All env-aware code takes `D1Database` as first param via `getDb(d1)` (matches `repeatOffers/repository`).
- Tests use `node:sqlite` D1 shim pattern from `apps/api/test/cross-border/e2e.test.ts:7-44`.
- Web uses `api.get/put` from `@/lib/api` (`apps/web/src/lib/api.ts`).
- No new npm deps.

```ts
import { and, eq, gte, inArray, desc, isNotNull } from 'drizzle-orm';
import { getDb } from '@vyro/db';
import { supplierBuyLeadSubscriptions, rfqs, rfqItems, products, users, supplierMembers } from '@vyro/db/schema';
import { newId } from '@vyro/shared';

export interface SubscriptionRow {
  supplierId: string;
  enabled: boolean;
  categoryIds: string[];
}

export interface RfqMatch {
  rfqId: string;
  rfqNumber: string;
  title: string;
  createdAt: number;
}

export interface EnabledSubRow {
  supplierId: string;
  recipientUserId: string;
  recipientEmail: string;
}

export const buyLeadsRepository = {
  async getSubscription(d1: D1Database, supplierId: string): Promise<SubscriptionRow | null> {
    const db = getDb(d1);
    const row = await db
      .select()
      .from(supplierBuyLeadSubscriptions)
      .where(eq(supplierBuyLeadSubscriptions.supplierId, supplierId))
      .get();
    if (!row) return null;
    return {
      supplierId: row.supplierId,
      enabled: row.enabled,
      categoryIds: JSON.parse(row.categoryIdsJson) as string[],
    };
  },

  async upsertSubscription(
    d1: D1Database,
    supplierId: string,
    enabled: boolean,
    categoryIds: string[],
  ): Promise<void> {
    const db = getDb(d1);
    const existing = await db
      .select({ id: supplierBuyLeadSubscriptions.id })
      .from(supplierBuyLeadSubscriptions)
      .where(eq(supplierBuyLeadSubscriptions.supplierId, supplierId))
      .get();
    const now = Date.now();
    const json = JSON.stringify(categoryIds);
    if (existing) {
      await db
        .update(supplierBuyLeadSubscriptions)
        .set({ enabled, categoryIdsJson: json, updatedAt: now })
        .where(eq(supplierBuyLeadSubscriptions.supplierId, supplierId));
    } else {
      await db.insert(supplierBuyLeadSubscriptions).values({
        id: newId(),
        supplierId,
        enabled,
        categoryIdsJson: json,
        createdAt: now,
        updatedAt: now,
      });
    }
  },

  async enabledSubscriptions(d1: D1Database): Promise<EnabledSubRow[]> {
    const db = getDb(d1);
    const rows = await db
      .select({
        supplierId: supplierBuyLeadSubscriptions.supplierId,
        userId: supplierMembers.userId,
        email: users.email,
      })
      .from(supplierBuyLeadSubscriptions)
      .innerJoin(supplierMembers, eq(supplierMembers.supplierId, supplierBuyLeadSubscriptions.supplierId))
      .innerJoin(users, eq(users.id, supplierMembers.userId))
      .where(and(eq(supplierBuyLeadSubscriptions.enabled, true), eq(supplierMembers.role, 'owner')));
    return rows.map((r) => ({
      supplierId: r.supplierId,
      recipientUserId: r.userId,
      recipientEmail: r.email,
    }));
  },

  async newRfqsMatchingCategories(
    d1: D1Database,
    categoryIds: string[],
    sinceMs: number,
    limit: number,
  ): Promise<RfqMatch[]> {
    if (categoryIds.length === 0) return [];
    const db = getDb(d1);
    const rows = await db
      .selectDistinct({
        rfqId: rfqs.id,
        rfqNumber: rfqs.rfqNumber,
        title: rfqs.title,
        createdAt: rfqs.createdAt,
      })
      .from(rfqs)
      .innerJoin(rfqItems, eq(rfqItems.rfqId, rfqs.id))
      .innerJoin(products, eq(products.id, rfqItems.productId))
      .where(and(
        eq(rfqs.status, 'open'),
        gte(rfqs.createdAt, sinceMs),
        isNotNull(rfqs.publishedAt),
        inArray(products.categoryId, categoryIds),
      ))
      .orderBy(desc(rfqs.createdAt))
      .limit(limit);
    return rows.map((r) => ({
      rfqId: r.rfqId,
      rfqNumber: r.rfqNumber,
      title: r.title,
      createdAt: r.createdAt,
    }));
  },
};
```

NOTE: Verify exact column names on `rfqs`, `rfqItems`, `products`, `supplierMembers`, `users` against the actual schema files before committing. Recon shows `rfqs.status`, `rfqs.publishedAt`, `rfqs.createdAt` exist; `rfqItems.productId` and `products.categoryId` should exist.
```ts
import type { BuyLeadsSubscription } from '@vyro/validation';
import { isFeatureEnabled } from '../../lib/featureFlags';
import { buyLeadsRepository } from './repository';
import { buyLeadsMatcher } from './matcher';

const FLAG = 'BUYLEADS_ENABLED';
const DIGEST_LIMIT = 10;
const WINDOW_MS = 24 * 60 * 60 * 1000;

function buildDigestEmail(
  recipientUserId: string,
  recipientEmail: string,
  matches: Array<{ rfqId: string; rfqNumber: string; title: string }>,
): { kind: 'buyleads_digest'; recipientUserId: string; recipientEmail: string; subject: string; body: string; link: string } {
  const subject = `${matches.length} new RFQs matching your categories`;
  const list = matches
    .map((m) => `<li><a href="/supplier/rfqs/${m.rfqId}">${m.rfqNumber} — ${escapeHtml(m.title)}</a></li>`)
    .join('');
  const body = `<p>Hi,</p><p>${matches.length} new RFQ${matches.length === 1 ? '' : 's'} matched your subscribed categories:</p><ul>${list}</ul><p>— Vyro BuyLeads</p>`;
  return { kind: 'buyleads_digest', recipientUserId, recipientEmail, subject, body, link: '/supplier/buyleads' };
}

function escapeHtml(s: string): string {
  return s.replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]!));
}

export const buyLeads = {
  async getMySubscription(d1: D1Database, supplierId: string): Promise<BuyLeadsSubscription> {
    const row = await buyLeadsRepository.getSubscription(d1, supplierId);
    if (!row) return { enabled: false, categoryIds: [] };
    return { enabled: row.enabled, categoryIds: row.categoryIds };
  },

  async updateMySubscription(
    d1: D1Database,
    supplierId: string,
    body: BuyLeadsSubscription,
  ): Promise<BuyLeadsSubscription> {
    await buyLeadsRepository.upsertSubscription(d1, supplierId, body.enabled, body.categoryIds);
    return body;
  },

  async runDailyDigest(
    env: { DB: D1Database; NOTIFICATIONS_QUEUE: { send: (m: unknown) => Promise<unknown> } },
    d1: D1Database,
  ): Promise<{ suppliersEmailed: number; rfqsSent: number }> {
    if (!(await isFeatureEnabled(d1, FLAG))) {
      return { suppliersEmailed: 0, rfqsSent: 0 };
    }
    const since = Date.now() - WINDOW_MS;
    const subs = await buyLeadsRepository.enabledSubscriptions(d1);
    let suppliersEmailed = 0;
    let rfqsSent = 0;
    for (const sub of subs) {
      const matches = await buyLeadsMatcher.topMatchesForSupplier(d1, sub.supplierId, since, DIGEST_LIMIT);
      if (matches.length === 0) continue;
      const email = buildDigestEmail(sub.recipientUserId, sub.recipientEmail, matches);
      try {
        await env.NOTIFICATIONS_QUEUE.send(email);
        suppliersEmailed++;
        rfqsSent += matches.length;
      } catch {
        // Skip this supplier; metric emits upstream.
      }
    }
    return { suppliersEmailed, rfqsSent };
  },
};
```

```ts
const router = new Hono<{ Bindings: Env }>();

function ctxOf(c: { get(k: string): unknown }): Ctx {
  const ctx = c.get('ctx') as Ctx | undefined;
  if (!ctx) throw httpError(401, 'UNAUTHORIZED', 'No session');
  return ctx;
}

async function ensureEnabled(d1: D1Database): Promise<void> {
  if (!(await isFeatureEnabled(d1, 'BUYLEADS_ENABLED'))) {
    throw httpError(404, 'NOT_FOUND', 'feature not enabled');
  }
}

function requireSupplier(c: { req: { query: (k: string) => string | undefined }; get: (k: string) => unknown }): { supplierId: string } {
  const ctx = ctxOf(c);
  const supplierId = c.req.query('supplierId');
```

```ts
function makeD1(sqlite: DatabaseSync): D1Database {
  const wrap = (sqlText: string, bound: unknown[] = []): D1PreparedStatement => {
    const st = {
      bind(...params: unknown[]) { return wrap(sqlText, [...bound, ...params]); },
      first: async (col?: string) => {
        const row = sqlite.prepare(sqlText).get(...bound) as Record<string, unknown> | undefined;
        return col ? (row?.[col] ?? null) : (row ?? null);
      },
      all: async () => ({ results: sqlite.prepare(sqlText).all(...bound), success: true, meta: {} }),
      run: async () => {
        const r = sqlite.prepare(sqlText).run(...bound) as unknown as { changes: unknown; lastInsertRowid: unknown };
        return { success: true, meta: { changes: r.changes, last_row_id: r.lastInsertRowid } };
      },
      raw: async () => (sqlite.prepare(sqlText).all(...bound) as Record<string, unknown>[]).map((r) => Object.values(r)),
    };
    return st as unknown as D1PreparedStatement;
  };
  return {
    prepare: (sqlText: string) => wrap(sqlText),
    exec: async (sqlText: string) => { sqlite.exec(sqlText); },
    batch: async (stmts: D1PreparedStatement[]) => {
      const out = [];
      for (const s of stmts) out.push(await (s as unknown as { run(): Promise<unknown> }).run());
      return out as never;
    },
  } as unknown as D1Database;
}
```

```tsx
import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api';
import type { BuyLeadsSubscription } from '@vyro/validation';

interface Category {
  id: string;
  slug: string;
  name: string;
  active: boolean;
}

export function BuyLeadsSettingsSection({ supplierId }: { supplierId: string }) {
  const qc = useQueryClient();
  const sub = useQuery({
    queryKey: ['supplier', supplierId, 'buyleads-sub'],
    queryFn: () => api.get<BuyLeadsSubscription>(`/supplier/buyleads/subs?supplierId=${supplierId}`),
    retry: false,
  });
  const cats = useQuery({
    queryKey: ['categories'],
    queryFn: () => api.get<{ categories: Category[] }>('/categories'),
    retry: false,
  });
  const [draft, setDraft] = useState<BuyLeadsSubscription | null>(null);
  const value = draft ?? sub.data ?? { enabled: false, categoryIds: [] };

  const save = useMutation({
    mutationFn: (next: BuyLeadsSubscription) =>
      api.put<BuyLeadsSubscription>(`/supplier/buyleads/subs?supplierId=${supplierId}`, next),
    onSuccess: (data) => {
      qc.setQueryData(['supplier', supplierId, 'buyleads-sub'], data);
    },
  });

  function toggle(id: string) {
    const set = new Set(value.categoryIds);
    if (set.has(id)) set.delete(id); else set.add(id);
    const next = { ...value, categoryIds: Array.from(set) };
    setDraft(next);
  }

  if (sub.isError) {
    return (
      <div className="text-sm text-ink-3">
        BuyLeads is not enabled for your account. Ask an admin to enable it.
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <label className="flex items-center gap-2 text-sm">
        <input
          type="checkbox"
          checked={value.enabled}
          onChange={(e) => { const next = { ...value, enabled: e.target.checked }; setDraft(next); }}
        />
        <span>Send me a daily email digest of new RFQs</span>
      </label>

      <div>
        <div className="text-sm font-semibold mb-2">Subscribed categories</div>
        <div className="grid grid-cols-2 gap-2">
          {(cats.data?.categories ?? []).filter((c) => c.active).map((c) => {
            const checked = value.categoryIds.includes(c.id);
            return (
              <label key={c.id} className="flex items-center gap-2 text-sm">
                <input
                  type="checkbox"
                  checked={checked}
                  onChange={() => toggle(c.id)}
                />
                <span>{c.name}</span>
              </label>
            );
          })}
        </div>
      </div>

      <button
        type="button"
        disabled={save.isPending || !draft}
        onClick={() => draft && save.mutate(draft)}
        className="rounded-md bg-ink px-3 py-1.5 text-sm text-paper disabled:opacity-50"
      >
        {save.isPending ? 'Saving…' : 'Save'}
      </button>
    </div>
  );
}
```
